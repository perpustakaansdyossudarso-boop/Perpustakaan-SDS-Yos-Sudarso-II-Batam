            const filtered = booksData.filter(book => {
                const matchesQuery = book.title.toLowerCase().includes(query) || book.author.toLowerCase().includes(query);
                const matchesCategory = category === 'all' || book.category === category;
                return matchesQuery && matchesCategory;
            });

            renderBooks(filtered);
        }

        function renderGallery(filter = 'all') {
            const grid = document.getElementById('gallery-grid');
            grid.innerHTML = '';

            const filtered = filter === 'all' ? galleryData : galleryData.filter(item => item.category === filter);

            filtered.forEach(item => {
                const div = document.createElement('div');
                div.className = "group relative rounded-2xl overflow-hidden shadow-md cursor-pointer h-52 bg-slate-200";
                div.onclick = () => openLightbox(item.url, item.title);
                div.innerHTML = `
                    <img src="${item.url}" alt="${item.title}" class="w-full h-full object-cover group-hover:scale-110 transition-transform duration-500" onerror="this.src='https://placehold.co/800x600/0284c7/ffffff?text=Foto+Galeri'">
                    <div class="absolute inset-0 bg-gradient-to-t from-black/80 via-black/20 to-transparent opacity-0 group-hover:opacity-100 transition-opacity p-4 flex flex-col justify-end text-white">
                        <p class="text-[10px] font-semibold text-brand-yellow uppercase">${item.category}</p>
                        <p class="font-heading font-bold text-xs">${item.title}</p>
                    </div>
                `;
                grid.appendChild(div);
            });
        }

        function filterGallery(cat) {
            document.querySelectorAll('.gallery-btn').forEach(btn => {
                btn.classList.remove('bg-brand-blue', 'text-white');
                btn.classList.add('bg-white', 'text-slate-700');
            });
            event.target.classList.remove('bg-white', 'text-slate-700');
            event.target.classList.add('bg-brand-blue', 'text-white');

            renderGallery(cat);
        }

        function renderCalendarAndAgenda() {
            const daysContainer = document.getElementById('calendar-days');
            daysContainer.innerHTML = '';
            for(let i=1; i<=31; i++) {
                const dayDiv = document.createElement('div');
                const isToday = i === 22; 
                const hasEvent = [28, 5].includes(i);

                dayDiv.className = `p-2 rounded-xl text-center relative ${isToday ? 'bg-brand-blue text-white font-bold' : 'hover:bg-slate-200 text-slate-700'}`;
                dayDiv.innerText = i;
                if (hasEvent && !isToday) {
                    dayDiv.innerHTML += `<span class="w-1.5 h-1.5 bg-brand-yellow rounded-full absolute bottom-1 left-1/2 -translate-x-1/2"></span>`;
                }
                daysContainer.appendChild(dayDiv);
            }

            const eventsList = document.getElementById('events-list');
            eventsList.innerHTML = '';
            agendaData.forEach(event => {
                const card = document.createElement('div');
                card.className = "bg-slate-50 p-4 rounded-2xl border border-slate-200 flex gap-4 items-center";
                card.innerHTML = `
                    <div class="p-3 bg-amber-100 text-brand-amber rounded-2xl text-center min-w-[55px]">
                        <i class="fa-regular fa-calendar text-lg block"></i>
                    </div>
                    <div>
                        <span class="text-xs font-bold text-brand-blue">${event.date} • ${event.location}</span>
                        <h4 class="font-heading font-bold text-slate-800 text-sm">${event.title}</h4>
                        <p class="text-xs text-slate-500 mt-0.5">${event.desc}</p>
                    </div>
                `;
                eventsList.appendChild(card);
            });
        }

        function switchAdminTab(tab) {
            document.getElementById('form-add-book').classList.add('hidden');
            document.getElementById('form-add-gallery').classList.add('hidden');
            document.getElementById('form-add-agenda').classList.add('hidden');

            document.querySelectorAll('[id^="tab-btn-"]').forEach(btn => {
                btn.classList.remove('border-brand-blue', 'text-brand-blue');
                btn.classList.add('border-transparent', 'text-slate-500');
            });

            if(tab === 'buku') {
                document.getElementById('form-add-book').classList.remove('hidden');
                document.getElementById('tab-btn-buku').classList.add('border-brand-blue', 'text-brand-blue');
            } else if(tab === 'galeri') {
                document.getElementById('form-add-gallery').classList.remove('hidden');
                document.getElementById('tab-btn-galeri').classList.add('border-brand-blue', 'text-brand-blue');
            } else {
                document.getElementById('form-add-agenda').classList.remove('hidden');
                document.getElementById('tab-btn-agenda').classList.add('border-brand-blue', 'text-brand-blue');
            }
        }
